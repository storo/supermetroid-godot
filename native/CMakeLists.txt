cmake_minimum_required(VERSION 3.20)
project(super_metroid_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
add_subdirectory(../third_party/godot-cpp godot-cpp)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(REFERENCE "${CMAKE_CURRENT_SOURCE_DIR}/../tools/reference_native/src")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/core/prepare.py" "${REFERENCE}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
file(GLOB BANKS "${REFERENCE}/sm_*.c")
list(REMOVE_ITEM BANKS "${REFERENCE}/sm_cpu_infra.c" "${REFERENCE}/sm_rtl.c")
file(GLOB DEVICES "${REFERENCE}/snes/*.c")
list(REMOVE_ITEM DEVICES "${REFERENCE}/snes/cpu.c" "${REFERENCE}/snes/spc.c")
add_library(sm_gameplay STATIC ${BANKS} ${DEVICES} "${REFERENCE}/util.c" "${REFERENCE}/spc_player.c" "${CMAKE_CURRENT_BINARY_DIR}/generated/sm_rtl.c" core/host.c core/devices.c)
target_include_directories(sm_gameplay PRIVATE "${REFERENCE}")
target_compile_options(sm_gameplay PRIVATE -O2 -fno-strict-aliasing -Wno-unknown-pragmas -Wno-unused-value)
add_library(sm_native SHARED bridge.cpp)
target_link_libraries(sm_native PRIVATE sm_gameplay godot-cpp)
set_target_properties(sm_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin" PREFIX "lib")
add_executable(sm_campaign_route EXCLUDE_FROM_ALL ../tools/native_probe/campaign_route.c)
target_include_directories(sm_campaign_route PRIVATE core)
target_link_libraries(sm_campaign_route PRIVATE sm_gameplay)
add_executable(sm_zebes_route EXCLUDE_FROM_ALL ../tools/native_probe/zebes_route.c)
target_include_directories(sm_zebes_route PRIVATE core)
target_link_libraries(sm_zebes_route PRIVATE sm_gameplay)
add_executable(sm_missile_route EXCLUDE_FROM_ALL ../tools/native_probe/missile_route.c)
target_include_directories(sm_missile_route PRIVATE core)
target_link_libraries(sm_missile_route PRIVATE sm_gameplay)
add_executable(sm_awaken_route EXCLUDE_FROM_ALL ../tools/native_probe/awaken_route.c)
target_include_directories(sm_awaken_route PRIVATE core)
target_link_libraries(sm_awaken_route PRIVATE sm_gameplay)
add_executable(sm_bomb_route EXCLUDE_FROM_ALL ../tools/native_probe/bomb_route.c)
target_include_directories(sm_bomb_route PRIVATE core)
target_link_libraries(sm_bomb_route PRIVATE sm_gameplay)
option(SM_BUILD_RENDER_ORACLE "Build offline pixel comparison tool (not part of gameplay)" OFF)
if(SM_BUILD_RENDER_ORACLE)
  add_executable(sm_raster_oracle ../tools/native_probe/raster_oracle.c)
  target_include_directories(sm_raster_oracle PRIVATE core "${REFERENCE}")
  target_link_libraries(sm_raster_oracle PRIVATE sm_gameplay)
endif()
